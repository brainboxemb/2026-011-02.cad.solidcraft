A simple printable bench dog inferred from the Mini-MFT reference image: cylindrical working section, small stop rim and a slightly oversized insertion stem split by two perpendicular slots. The four stem quadrants flex inward for a passive friction fit; there is no screw mechanism.

Construction stages

  1. Cylinder — Ø20 × 35 mm working section
  2. Cylinder — join Ø24 × 3 mm stop rim
  3. Cylinder — join Ø20.2 × 38 mm insertion stem
  4. Box cut — first 1.5 × 30 mm flex slot
  5. Box cut — second perpendicular flex slot
  6. Four flexible stem quadrants provide passive friction

"""Reusable 20 mm flex bench dog based on the supplied Mini-MFT reference image.

The fit is intentionally passive: a slightly oversized insertion stem is split
by two crossing slots so the four quadrants can flex inward in a 20 mm dog hole.
No screw, nut, O-ring or separate expansion part is used.
"""


@component
class FlexBenchDog:
    # Native fit parameters. These are the dimensions most useful to tune after
    # a first print in a real 20 mm worktop.
    hole_diameter = 20.0
    interference = 0.20
    rim_diameter = 24.0
    slot_width = 1.5

    # Fixed first-prototype proportions inferred from the supplied image.
    WORKING_HEIGHT = 35.0
    RIM_THICKNESS = 3.0
    INSERTION_LENGTH = 38.0
    SLOT_LENGTH = 30.0

    def build(self):
        dog = Cylinder(
            d=self.hole_diameter,
            h=self.WORKING_HEIGHT,
            anchor=Anchor.BOTTOM,
        )

        rim = Cylinder(
            d=self.rim_diameter,
            h=self.RIM_THICKNESS,
            anchor=Anchor.BOTTOM,
        ).at(0, 0, self.WORKING_HEIGHT, anchor=Anchor.BOTTOM)
        dog = dog + rim

        stem_diameter = self.hole_diameter + self.interference
        stem_z = self.WORKING_HEIGHT + self.RIM_THICKNESS
        stem = Cylinder(
            d=stem_diameter,
            h=self.INSERTION_LENGTH,
            anchor=Anchor.BOTTOM,
        ).at(0, 0, stem_z, anchor=Anchor.BOTTOM)
        dog = dog + stem

        slot_z = stem_z + self.INSERTION_LENGTH - self.SLOT_LENGTH
        slot_x = Box(
            stem_diameter + 2.0,
            self.slot_width,
            self.SLOT_LENGTH,
        ).at(0, 0, slot_z, anchor=Anchor.BOTTOM)
        slot_y = Box(
            self.slot_width,
            stem_diameter + 2.0,
            self.SLOT_LENGTH,
        ).at(0, 0, slot_z, anchor=Anchor.BOTTOM)

        dog = dog - slot_x
        dog = dog - slot_y
        return dog

"""FlexBenchDog expressed directly as SolidCraft commands."""

from scpc import Script

script = Script()

for name, value in (
    ("hole_diameter", 20.0),
    ("interference", 0.20),
    ("rim_diameter", 24.0),
    ("slot_width", 1.5),
):
    script.add("parameters.add", name=name, expression=str(value), unit="mm")

script.add(
    "solid.cylinder",
    height=35,
    diameter="hole_diameter",
    base=[0, 0, 0],
    axis="Z",
    operation="new",
    body_name="FlexBenchDog",
)
script.add(
    "solid.cylinder",
    height=3,
    diameter="rim_diameter",
    base=[0, 0, 35],
    axis="Z",
    operation="join",
    targets=["FlexBenchDog"],
)
script.add(
    "solid.cylinder",
    height=38,
    diameter="hole_diameter + interference",
    base=[0, 0, 38],
    axis="Z",
    operation="join",
    targets=["FlexBenchDog"],
)
script.add(
    "solid.box",
    length="hole_diameter + interference + 2",
    width="slot_width",
    height=30,
    corner=[
        "-(hole_diameter + interference + 2) / 2",
        "-slot_width / 2",
        46,
    ],
    operation="cut",
    targets=["FlexBenchDog"],
)
script.add(
    "solid.box",
    length="slot_width",
    width="hole_diameter + interference + 2",
    height=30,
    corner=[
        "-slot_width / 2",
        "-(hole_diameter + interference + 2) / 2",
        46,
    ],
    operation="cut",
    targets=["FlexBenchDog"],
)
