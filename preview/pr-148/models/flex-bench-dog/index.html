FlexBenchDog

Passive 20 mm MFT bench dog with a four-segment flex stem

A simple printable bench dog inferred from the Mini-MFT reference image: 20 mm above and 20 mm below the worktop, with a nominal Ø20 insert and a small Ø20.2 flare only at the free tip. Two perpendicular slots create four flexible legs; a larger bottom chamfer provides an insertion lead-in. There is no screw mechanism.

Construction stages

  1. Cylinder — Ø24 × 3 mm stop rim at worktop datum
  2. Cylinder — join Ø20 × 17 mm working section to Z=20
  3. Cylinder — join Ø20 × 20 mm insertion below worktop
  4. Cylinder — add Ø20.2 × 2 mm flared free tip
  5. Two 1.5 × 16 mm crossing cuts create four flex legs
  6. Chamfer top and bottom edges, then cut a recessed square box logo

"""Reusable 20 mm flex bench dog based on the supplied Mini-MFT reference image.

The CAD orientation is functional rather than a prescribed print orientation:
the worktop surface is Z=0, the insert extends 20 mm downward and the dog rises
20 mm upward. A small flare at the free insert end provides passive spring
preload after the stem is split into four flexible quadrants.
"""


@component
class FlexBenchDog:
    hole_diameter = 20.0
    interference = 0.20
    rim_diameter = 24.0
    slot_width = 1.5

    ABOVE_WORKTOP = 20.0
    RIM_THICKNESS = 3.0
    INSERTION_LENGTH = 20.0
    TIP_FLARE_LENGTH = 2.0
    SLOT_LENGTH = 16.0
    TOP_CHAMFER = 0.6
    BOTTOM_CHAMFER = 1.0
    LOGO_SIZE = 8.0
    LOGO_DEPTH = 0.6

    def build(self):
        # Rim bottom is the worktop datum. The narrower working cylinder starts
        # on the rim and ends at 20 mm above the worktop.
        rim = Cylinder(
            d=self.rim_diameter,
            h=self.RIM_THICKNESS,
            anchor=Anchor.BOTTOM,
        )

        working_height = self.ABOVE_WORKTOP - self.RIM_THICKNESS
        upper = Cylinder(
            d=self.hole_diameter,
            h=working_height,
            anchor=Anchor.BOTTOM,
        ).at(0, 0, self.RIM_THICKNESS, anchor=Anchor.BOTTOM)
        dog = rim + upper

        # Straight insert root below Z=0, with only the free 2 mm flared.
        insert = Cylinder(
            d=self.hole_diameter,
            h=self.INSERTION_LENGTH,
            anchor=Anchor.BOTTOM,
        ).at(0, 0, -self.INSERTION_LENGTH, anchor=Anchor.BOTTOM)
        dog = dog + insert

        flare = Cylinder(
            d=self.hole_diameter + self.interference,
            h=self.TIP_FLARE_LENGTH,
            anchor=Anchor.BOTTOM,
        ).at(0, 0, -self.INSERTION_LENGTH, anchor=Anchor.BOTTOM)
        dog = dog + flare

        # Give the complete free insert edge a larger lead-in before splitting
        # it into four flexible legs.
        dog = dog.chamfer(
            self.BOTTOM_CHAMFER,
            edges=[(
                (self.hole_diameter + self.interference) / 2,
                0,
                -self.INSERTION_LENGTH,
            )],
        )

        # Two native sketch cuts form the cross slots. They start at the free
        # end and stop 4 mm below the worktop, leaving a solid flex root.
        slot_span = self.hole_diameter + self.interference + 2.0

        with Sketch(
            name="Flex slot X",
            on=dog,
            plane="XY",
            origin=(0, 0, -self.INSERTION_LENGTH),
        ) as slot_x:
            Rectangle2D(slot_span, self.slot_width)
        dog = dog - slot_x.extrude(self.SLOT_LENGTH)

        with Sketch(
            name="Flex slot Y",
            on=dog,
            plane="XY",
            origin=(0, 0, -self.INSERTION_LENGTH),
        ) as slot_y:
            Rectangle2D(self.slot_width, slot_span)
        dog = dog - slot_y.extrude(self.SLOT_LENGTH)

        # Light chamfer on the visible upper perimeter.
        dog = dog.chamfer(
            self.TOP_CHAMFER,
            edges=[(self.hole_diameter / 2, 0, self.ABOVE_WORKTOP)],
        )

        # Small recessed square used as a simple BBE/box mark.
        with Sketch(
            name="Top logo",
            on=dog,
            plane="XY",
            origin=(0, 0, self.ABOVE_WORKTOP),
        ) as logo:
            Rectangle2D(self.LOGO_SIZE, self.LOGO_SIZE)
        dog = dog - logo.extrude(-self.LOGO_DEPTH)

        return dog

"""FlexBenchDog expressed directly as SolidCraft commands."""

from scpc import Script

script = Script()

for name, value in (
    ("hole_diameter", 20.0),
    ("interference", 0.20),
    ("rim_diameter", 24.0),
    ("slot_width", 1.5),
):
    script.add("parameters.add", name=name, expression=str(value), unit="mm")

# Rim at the worktop datum.
script.add(
    "solid.cylinder",
    height=3,
    diameter="rim_diameter",
    base=[0, 0, 0],
    axis="Z",
    operation="new",
    body_name="FlexBenchDog",
)

# Working section: top of rim to 20 mm above the worktop.
script.add(
    "solid.cylinder",
    height=17,
    diameter="hole_diameter",
    base=[0, 0, 3],
    axis="Z",
    operation="join",
    targets=["FlexBenchDog"],
)

# Straight 20 mm insert and a slightly oversized free tip.
script.add(
    "solid.cylinder",
    height=20,
    diameter="hole_diameter",
    base=[0, 0, -20],
    axis="Z",
    operation="join",
    targets=["FlexBenchDog"],
)
script.add(
    "solid.cylinder",
    height=2,
    diameter="hole_diameter + interference",
    base=[0, 0, -20],
    axis="Z",
    operation="join",
    targets=["FlexBenchDog"],
)

# Larger lead-in chamfer on the free insert edge.
script.add(
    "solid.chamfer",
    body="FlexBenchDog",
    distance=1.0,
    edges=[[10.1, 0, -20]],
)

# Flex slot X.
script.add(
    "sketch.create",
    plane={"origin": [0, 0, -20], "x_dir": [1, 0, 0], "y_dir": [0, 1, 0]},
    name="Flex slot X",
)
script.add(
    "sketch.rectangle.two_point",
    p0=["-(hole_diameter + interference + 2) / 2", "-slot_width / 2"],
    p1=["(hole_diameter + interference + 2) / 2", "slot_width / 2"],
)
script.add("sketch.finish")
script.add(
    "solid.extrude",
    distance=16,
    direction="positive",
    operation="cut",
    targets=["FlexBenchDog"],
    name="Flex slot X",
)

# Flex slot Y.
script.add(
    "sketch.create",
    plane={"origin": [0, 0, -20], "x_dir": [1, 0, 0], "y_dir": [0, 1, 0]},
    name="Flex slot Y",
)
script.add(
    "sketch.rectangle.two_point",
    p0=["-slot_width / 2", "-(hole_diameter + interference + 2) / 2"],
    p1=["slot_width / 2", "(hole_diameter + interference + 2) / 2"],
)
script.add("sketch.finish")
script.add(
    "solid.extrude",
    distance=16,
    direction="positive",
    operation="cut",
    targets=["FlexBenchDog"],
    name="Flex slot Y",
)

# Light chamfer on the visible upper perimeter.
script.add(
    "solid.chamfer",
    body="FlexBenchDog",
    distance=0.6,
    edges=[[10, 0, 20]],
)

# Recessed square box logo.
script.add(
    "sketch.create",
    plane={"origin": [0, 0, 20], "x_dir": [1, 0, 0], "y_dir": [0, 1, 0]},
    name="Top logo",
)
script.add(
    "sketch.rectangle.two_point",
    p0=[-4, -4],
    p1=[4, 4],
)
script.add("sketch.finish")
script.add(
    "solid.extrude",
    distance=0.6,
    direction="negative",
    operation="cut",
    targets=["FlexBenchDog"],
    name="Top logo",
)
