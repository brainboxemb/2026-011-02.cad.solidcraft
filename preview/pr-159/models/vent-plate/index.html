Python model

"""Vent Plate from the SolidCraft example, using one native feature pattern.

The source cut is authored once. SCPF 0.6.1 lowers rectangular_pattern()
directly to SolidCraft's solid.pattern.rectangular history feature.
"""


class Plate:
    cols = 6
    rows = 3
    pitch = 16.0

    def build(self):
        # Match the source SolidCraft box corner [0, 0, 0].
        plate = Box(120, 80, 4).at(
            60, 40, 0,
            anchor=Anchor.BOTTOM,
        )
        plate = plate.fillet(8, edges=Edges.VERTICAL)

        # The source rectangle is X=18..24, Y=12..26. The attached sketch
        # coordinates are relative to the 120 x 80 plate centre (60, 40).
        with Sketch(name="Vent", on=plate, parent=Anchor.BOTTOM) as vent:
            vent.add(
                Rectangle2D(6, 14),
                offset=(-39, -21),
            )

        source_cut = vent.extrude(4)
        plate = plate - source_cut

        return plate.rectangular_pattern(
            source_cut,
            count=self.cols,
            spacing=self.pitch,
            direction=Axis.X,
            count2=self.rows,
            spacing2=22,
            direction2=Axis.Y,
        )

"""Vent Plate from the official SolidCraft example, expressed as native commands.

This intentionally mirrors the source feature history:
Box1 -> Fillet1 -> Slot -> Vent -> RectangularPattern1.

This file remains independent from the SCPF model and keeps the source
SolidCraft names, including the separate Slot sketch and Vent cut feature.
"""
from scpc import Script


script = Script()

script.add("parameters.add", name="cols", expression="6", unit="")
script.add("parameters.add", name="rows", expression="3", unit="")
script.add(
    "parameters.add",
    name="pitch",
    expression="16",
    unit="mm",
    comment="slot spacing",
)

script.add(
    "solid.box",
    length=120,
    width=80,
    height=4,
    corner=[0, 0, 0],
    body_name="Plate",
)

script.add(
    "solid.fillet",
    body="Plate",
    radius=8,
    edges=[
        [0, 0, 2],
        [120, 0, 2],
        [120, 80, 2],
        [0, 80, 2],
    ],
)

script.add("sketch.create", plane="XY", name="Slot")
script.add(
    "sketch.rectangle.two_point",
    p0=[18, 12],
    p1=[24, 26],
)
script.add("sketch.finish")

script.add(
    "solid.extrude",
    distance=4,
    direction="positive",
    operation="cut",
    targets=["Plate"],
    name="Vent",
)

script.add(
    "solid.pattern.rectangular",
    features=["Vent"],
    dir1=[1, 0, 0],
    count1="cols",
    spacing1="pitch",
    dir2=[0, 1, 0],
    count2="rows",
    spacing2=22,
    name="RectangularPattern1",
)
