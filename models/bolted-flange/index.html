Construction stages

  1. Disc — Ø100 Sketch on XY, extrude 10 mm
  2. Hub — Ø44 top-face Sketch, join 16 mm
  3. Bore — central Ø24 cut through the 26 mm overall height
  4. BoltHole — one Ø9 source cut at pcd / 2
  5. CircularPattern — bolts instances around the Z axis

Python model

"""Bolted flange from the official SolidCraft sample.

The model keeps one native source bolt-hole cut and repeats that feature with
one native circular history pattern. SCPF uses one name for an explicit Sketch
and its extrusion, so the source Sketch named "Bolt" is represented here as
"BoltHole"; SCPC keeps the original separate names exactly.
"""


class Flange:
    bolts = 6
    pcd = 70.0

    def build(self):
        with Sketch(name="Disc", plane="XY") as disc:
            Circle2D(d=100)
        flange = disc.extrude(10)

        with Sketch(name="Hub", on=flange, parent=Anchor.TOP) as hub:
            Circle2D(d=44)
        hub_feature = hub.extrude(16)
        flange = flange + hub_feature

        with Sketch(
            name="Bore",
            on=flange,
            plane="XY",
            origin=(0, 0, 0),
        ) as bore:
            Circle2D(d=24)
        bore_cut = bore.extrude(26)
        flange = flange - bore_cut

        with Sketch(
            name="BoltHole",
            on=flange,
            plane="XY",
            origin=(0, 0, 0),
        ) as bolt:
            bolt.add(
                Circle2D(d=9),
                offset=(self.pcd / 2, 0),
            )
        bolt_cut = bolt.extrude(10)
        flange = flange - bolt_cut

        return flange.circular_pattern(
            bolt_cut,
            axis=Axis.Z,
            count=self.bolts,
        )

"""Official SolidCraft bolted-flange sample expressed as SCPC commands.

This intentionally preserves the source command history and names:
Disc -> Hub -> Bore -> Bolt/BoltHole -> CircularPattern.
"""
from scpc import Script


script = Script()

script.add(
    "parameters.change",
    name="bolts",
    expression="6",
    unit="",
    comment="number of bolt holes",
)
script.add(
    "parameters.change",
    name="pcd",
    expression="70 mm",
    comment="bolt circle diameter",
)

script.add("sketch.create", plane="XY", name="Disc")
script.add("sketch.circle.center", center=[0, 0], diameter=100)
script.add("sketch.finish")
script.add(
    "solid.extrude",
    distance=10,
    body_name="Flange",
    name="Disc",
)

script.add(
    "sketch.create",
    plane={"face": [0, 0, 10]},
    name="Hub",
)
script.add("sketch.circle.center", center=[0, 0], diameter=44)
script.add("sketch.finish")
script.add(
    "solid.extrude",
    distance=16,
    operation="join",
    name="Hub",
)

script.add("sketch.create", plane="XY", name="Bore")
script.add("sketch.circle.center", center=[0, 0], diameter=24)
script.add("sketch.finish")
script.add(
    "solid.extrude",
    distance=26,
    operation="cut",
    name="Bore",
)

script.add("sketch.create", plane="XY", name="Bolt")
script.add(
    "sketch.circle.center",
    center=[35, 0],
    diameter=9,
)
script.add("sketch.finish")
script.add(
    "solid.extrude",
    distance=10,
    operation="cut",
    name="BoltHole",
)

script.add(
    "solid.pattern.circular",
    features=["BoltHole"],
    axis="Z",
    count="bolts",
)
