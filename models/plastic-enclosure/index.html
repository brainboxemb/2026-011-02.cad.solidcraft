Construction stages

  1. Enclosure footprint — sketch the 80 × 50 mm base
  2. Extrude1 — create the 30 mm solid
  3. Shell1 — hollow to 2 mm with the top open
  4. Lip1 — add a 1 × 2 mm rim lip
  5. SnapFit1 — add the 14 mm cantilever snap
  6. Boss1 / Boss2 — add two Ø7 × 12 mm screw bosses
  7. Rest1 — add the 16 × 24 × 3 mm centre rest

Python model

class Enclosure:
    """Rebuild the SolidCraft plastic enclosure with native plastic features."""

    length = 80.0
    width = 50.0
    height = 30.0
    wall = 2.0

    def build(self):
        enclosure = Box(self.length, self.width, self.height, anchor=Anchor.BOTTOM)
        enclosure = enclosure.shell(self.wall, open=Faces.TOP)

        lip = Lip(width=1, height=2).attach(
            enclosure,
            parent=Anchor.TOP,
            offset=(X.MIN + 1, Y.CENTER),
        )
        enclosure = enclosure + lip

        snap = SnapFit(
            length=14,
            thickness=1.5,
            width=6,
            catch_depth=1,
            catch_length=2,
            hook=(0, -1, 0),
        ).attach(
            enclosure,
            parent=Anchor.INNER_BOTTOM,
            offset=(X.CENTER, Y.MAX - 6),
        )
        enclosure = enclosure + snap

        boss1 = Boss(d=7, h=12, hole_d=2.5).attach(
            enclosure,
            parent=Anchor.INNER_BOTTOM,
            offset=(X.MIN + 10, Y.MIN + 10),
        )
        enclosure = enclosure + boss1

        boss2 = Boss(d=7, h=12, hole_d=2.5).attach(
            enclosure,
            parent=Anchor.INNER_BOTTOM,
            offset=(X.MAX - 10, Y.MAX - 10),
        )
        enclosure = enclosure + boss2

        rest = Rest(width=16, length=24, height=3).attach(
            enclosure,
            parent=Anchor.INNER_BOTTOM,
        )
        return enclosure + rest
